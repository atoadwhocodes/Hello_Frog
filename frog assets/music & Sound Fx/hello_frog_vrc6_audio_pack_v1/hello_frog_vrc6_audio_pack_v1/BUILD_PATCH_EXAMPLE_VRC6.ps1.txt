; Example build patch for a split-object VRC6 audio path
; adjust paths/output names to your repo layout

ca65 hello-pond.asm -o build/hello-pond.o
ca65 audio/audio_vrc6.asm -o build/audio_vrc6.o
ca65 audio/audio_sfx_data_vrc6.asm -o build/audio_sfx_data_vrc6.o
ca65 audio/audio_music_data_vrc6.asm -o build/audio_music_data_vrc6.o

ld65 -C nes.cfg ^
  build/hello-pond.o ^
  build/audio_vrc6.o ^
  build/audio_sfx_data_vrc6.o ^
  build/audio_music_data_vrc6.o ^
  -o build/hello-pond-vrc6.nes
